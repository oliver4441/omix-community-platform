Good catch, Alex. We definitely need memoization there. I drafted a quick wrapper we could use for the complex selectors to prevent those unnecessary render cycles.

typescript 

import { useMemo } from 'react';

export const useDeepSelector = (selector) => {
  const state = useAppState();
  
  return useMemo(() => {
    return selector(state);
  }, [state, selector]);
};
